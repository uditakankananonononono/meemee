"""Both record and source visibility gate context assembled for an audience."""

import pytest
from test_personal_context_backends import stores  # noqa: F401

from meemee.context import ContextRecord


@pytest.mark.parametrize('reader', ['recent', 'search', 'reflection'])
def test_source_audience_restricts_record_retrieval(stores, reader):  # noqa: F811
    _, store = stores()
    store.register_source('owner', 'source', 'test', {}, permission='shared')
    store.ingest(ContextRecord('owner', 'source', 'record', 'event', 'title', 'fixture', '2026-10-08', {}))
    if reader == 'recent':
        records = store.recent('owner')
    elif reader == 'search':
        records = store.search('owner', 'fixture')
    else:
        records = store.reflection_batch('owner', 0, store.owner_watermarks()['owner'])
    assert records == []
    # Updating source permission takes effect on already published evidence.
    store.register_source('owner', 'source', 'test', {}, permission='private')
    assert len(store.recent('owner')) == 1
    assert len(store.search('owner', 'fixture')) == 1
    assert len(store.reflection_batch('owner', 0, store.owner_watermarks()['owner'])) == 1


def test_private_source_not_exposed_by_shared_record(stores):  # noqa: F811
    _, store = stores()
    store.register_source('owner', 'source', 'test', {}, permission='private')
    store.ingest(ContextRecord('owner', 'source', 'record', 'event', 'title', 'fixture',
                               '2026-10-08', {}, visibility='shared'))
    assert store.recent('owner', allowed={'shared'}) == []
    assert store.search('owner', 'fixture', allowed={'shared'}) == []
